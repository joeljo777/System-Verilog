class driver;
  
  int no_transactions;
  
  virtual intf vif; //virtual interface are pointer to actual interface
  
  
  mailbox gen2driv;
  
  function new(virtual intf vif, mailbox gen2driv);
    //this keyword is used to represent calss variables
    this.vif = vif;
    this.gen2driv = gen2driv;
  endfunction 
  
  task reset;
    wait(vif.rst);
    $display("[driver]--------- reset started-------");
    vif.a <=0;
    vif.b <=0;
    vif.valid <=0;
    wait(!vif.rst);
    $display("[driver]---------reset ended----");
  endtask
  
  task main;
      transaction trans;
    forever begin //infinet loop
//       transaction trans;
      gen2driv.get(trans); //take values from mailbox 
      @(posedge vif.clk);
      vif.valid <=1;
      //packet to pin 
      vif.a <=trans.a;
      vif.b <= trans.b;
      @(posedge vif.clk);
      vif.valid <=0;
      trans.c <= vif.c;
      @(posedge vif.clk);
      trans.display("Driver");
      no_transactions++;
    end
  endtask
endclass
