class generator;
  
  rand transaction trans;
  
  int repeat_count;
  
  mailbox gen2driv; //communication bwt gen and driv
  
  event ended; //indicated the end of generating random values
  
  function new(mailbox gen2driv); //constructor
    
    this.gen2driv = gen2driv; //this keyword represents class variable
  endfunction 
  
  task main();
    repeat (repeat_count) begin
      trans = new(); 
      //trans.randomize() --> trigger to generate random values
      if(!trans.randomize()) $fatal ("gen:: trans randomize failed");
      trans.display("[generator]");
      gen2driv.put(trans); //storing the value in mailbox 
    end
    ->ended; //generator has done it operation
  endtask
endclass
