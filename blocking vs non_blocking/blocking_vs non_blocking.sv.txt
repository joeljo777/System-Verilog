//blocking
module blocking( input clk,d,q1,q2);
  always(@ posedge clk) begin
    q1= d;
    q2= q1;
  end
endmodule


//non_blocking
module nonblocking( input clk,d,q1,q2);
  always(@ posedge clk) begin
    q1<= d;
    q2<= q1;
  end
endmodule
