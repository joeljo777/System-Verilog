
// Blocking Up/Down Counter
module blocking_counter(
    input clk, rst, up_down,
    output reg [3:0] count
);

    always @(posedge clk) begin
        if (rst)
            count = 4'b0000;

        else if (up_down)
            count = count + 1'b1;

        else
            count = count - 1'b1;
    end

endmodule


// Non-Blocking Up/Down Counter
module nonblocking_counter(
    input clk, rst, up_down,
    output reg [3:0] count
);

    always @(posedge clk) begin
        if (rst)
            count <= 4'b0000;

        else if (up_down)
            count <= count + 1'b1;

        else
            count <= count - 1'b1;
    end

endmodule


//test_bench
module tb;

    reg clk, rst, up_down;

    wire [3:0] blocking_count;
    wire [3:0] nonblocking_count;

    // Module instantiation
    blocking_counter B(
        .clk(clk),
        .rst(rst),
        .up_down(up_down),
        .count(blocking_count)
    );

    nonblocking_counter NB(
        .clk(clk),
        .rst(rst),
        .up_down(up_down),
        .count(nonblocking_count)
    );

    // Clock generation
    always #5 clk = ~clk;

    initial begin

        $dumpfile("dump.vcd");
        $dumpvars(0, tb);

        clk = 0;
        rst = 1;
        up_down = 1;

        // Reset
        #10 rst = 0;

        // Up counting
        #60;

        // Down counting
        up_down = 0;
        #60;

        // Up counting again
        up_down = 1;
        #40;

        $finish;
    end

    initial begin
        $monitor("Time=%0t | Direction=%b | Blocking=%d | NonBlocking=%d",
                  $time, up_down, blocking_count, nonblocking_count);
    end

endmodule
