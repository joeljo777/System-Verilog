class scoreboard;
  mailbox mon2scb;
  int no_transactions;

  function new(mailbox mon2scb);
    this.mon2scb = mon2scb;
  endfunction

  task main;
    transaction trans;
    forever begin
      mon2scb.get(trans);
      if ({trans.cout, trans.sum} == (trans.a + trans.b + trans.cin)) begin
        $display("[scoreboard] Result matched: a=%0b + b=%0b + cin=%0b => {cout=%0b, sum=%0b}",
                 trans.a, trans.b, trans.cin, trans.cout, trans.sum);
      end else begin
        $error("[scoreboard] WRONG RESULT! expected: %0b actual: {cout=%0b, sum=%0b}",
               (trans.a + trans.b + trans.cin), trans.cout, trans.sum);
      end
      no_transactions++;
      trans.display("[scoreboard]");
    end
  endtask
endclass
